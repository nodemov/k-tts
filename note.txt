# ขั้นตอนทำหนังสือเสียง (รันจากโฟลเดอร์ k-tts ตามลำดับ)

# 1) ดึงตอนนิยายจาก DB -> chapters/0001.txt ...
#    (เลือกช่วงตอนได้: ใส่ <from> <to> ต่อท้าย เช่น ... 51 100)
go run ./cmd/fetch-chapters 01a195b5-eb36-79eb-af7a-aa4430d8b4f3

# 2) (ไม่บังคับ) ตรวจข้อความก่อนแปลงเป็นเสียง -> chapters-clean/
#    0051.txt       = ข้อความ clean ที่ TTS จะอ่านจริง
#    0051.parts.txt = แต่ละท่อนที่ส่งไป TTS (ตัดที่ช่องว่างเท่านั้น)
#    ขั้นที่ 3 ทำความสะอาดข้อความเองอยู่แล้ว ขั้นนี้มีไว้ดูผลเท่านั้น
go run ./cmd/clean-chapters -parts -engine router

# 3) แปลงข้อความเป็นเสียง -> output/0001.mp3 ...
#    ทำใหม่เฉพาะบางตอน: เพิ่ม -only 51-60 หรือ -only 102,218
#go run . -engine router -workers 12 -fallback=false

go run . -engine router -workers 12

# 4) รวมไฟล์เสียง ไฟล์ละ 50 ตอน -> output/combined/
go run ./cmd/combine-audio 50

# 5) ดูผลลัพธ์
cd output/combined && ls -la
